import pandas as pd
df = pd.read_excel('Wheat_RNAseq_from_multiple_studies.xlsx')

# Get S12 full cultivar list with FHB groups
s12 = df[df['Run'].str.startswith('SRR1467', na=False) | df['Run'].str.startswith('SRR1467', na=False)]
s12_full = df[(df['Run'] >= 'SRR14673823') & (df['Run'] <= 'SRR14674100')]
print(f"S12 samples: {len(s12_full)}")
print(s12_full.groupby('Fusarium_head_blight_resistance_group')['Run'].count())
print("\nSumai3 vs non:")
print(s12_full.groupby('Sumai3_versus_non_Sumai3_descendant')['Run'].count())

# Count by study
def get_study(run):
    if run.startswith('SRR856'): return 'S1'
    if run.startswith('SRR177'): return 'S2'
    if run.startswith('SRR862'): return 'S3'
    if run.startswith('SRR1942'): return 'S4'
    if run.startswith('SRR8309'): return 'S5'
    if run.startswith('SRR1316') or run.startswith('SRR1317'): return 'S6'
    if run == 'SRR639194': return 'S7_EXCLUDED'
    if run.startswith('SRR2096'): return 'S8'
    if run.startswith('ERR2275'): return 'S9'
    if run.startswith('SRR1058'): return 'S10'
    if run.startswith('SRR309'): return 'S11'
    if run >= 'SRR14673823' and run <= 'SRR14674100': return 'S12'
    if run.startswith('SRR1234'): return 'S11b'
    if run.startswith('SRR1702'): return 'S13'
    if run.startswith('SRR1695'): return 'S14'
    if run.startswith('ERR120'): return 'S15'
    return 'other'

df['study'] = df['Run'].apply(get_study)
summary = df.groupby('study').agg(
    n=('Run','count'),
    layout=('LibraryLayout', lambda x: '+'.join(sorted(x.unique()))),
    tissue=('tissue', lambda x: '/'.join(sorted(set(str(v) for v in x if str(v)!='nan')))[:70]),
    cultivar=('Cultivar', lambda x: str(len(x.unique())) + ' cultivars'),
    instrument=('Instrument', lambda x: '/'.join(sorted(x.unique()))[:60]),
).reset_index()
print("\n=== COMPLETE STUDY SUMMARY ===")
for _, r in summary.iterrows():
    print(f"  {r['study']:12} n={r['n']:3}  {r['layout']:6}  {r['instrument'][:40]:40}  tissue={r['tissue'][:40]}")
